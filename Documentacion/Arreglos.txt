      function toggleHostingPanel() {
        var panel = document.getElementById('hosting-panel');
        var dashboard = document.getElementById('main-dashboard-content');
        var sidebarItem = document.getElementById('sidebar-hosting-item');
        if (panel.style.display === 'none') {
          panel.style.display = 'block';
          dashboard.style.display = 'none';
          sidebarItem.classList.add('active');
          loadHostingConfig();
        } else {
          panel.style.display = 'none';
          dashboard.style.display = 'block';
          sidebarItem.classList.remove('active');
        }
      }
      function copyField(id) {
        var el = document.getElementById(id);
        if (el && el.value) {
          navigator.clipboard.writeText(el.value).then(function() {
            showToast('Copiado: ' + el.value);
          });
        }
      }
      function copyText(text) {
        navigator.clipboard.writeText(text).then(function() {
          showToast('Copiado: ' + text);
        });
      }
      function showToast(msg) {
        var t = document.createElement('div');
        t.textContent = msg;
        t.style.cssText = 'position:fixed;bottom:30px;right:30px;background:#7366ff;color:#fff;padding:10px 20px;border-radius:8px;z-index:99999;font-size:13px;box-shadow:0 4px 12px rgba(0,0,0,.2);';
        document.body.appendChild(t);
        setTimeout(function(){ document.body.removeChild(t); }, 2000);
      }
      function applyDomainConnect() {
        var domain = document.getElementById('f-domain-connect').value.trim();
        var dns1 = document.getElementById('f-dns1').value.trim();
        var dns2 = document.getElementById('f-dns2').value.trim();
        if (dns1) document.getElementById('dns1-display').innerText = dns1;
        if (dns2) document.getElementById('dns2-display').innerText = dns2;
        if (domain) {
          var godaddyBtn = document.getElementById('btn-godaddy');
          godaddyBtn.onclick = function(){ window.open('https://dcc.godaddy.com/control/portfolio/'+domain+'/settings', '_blank'); };
          var wpadmin = document.getElementById('btn-wpadmin');
          if(wpadmin) wpadmin.onclick = function(){ window.open('https://'+domain+'/wp-admin', '_blank'); };
          var plugins = document.getElementById('btn-plugins');
          if(plugins) plugins.onclick = function(){ window.open('https://'+domain+'/wp-admin/plugins.php', '_blank'); };
          var importBtn = document.getElementById('btn-import-wpress');
          if(importBtn) importBtn.onclick = function(){ window.open('https://'+domain+'/wp-admin/admin.php?page=all-in-one-wp-migration-import', '_blank'); };
          showToast('Dominio aplicado: ' + domain);
        }
      }
      function applyExportUrl() {
        var url = document.getElementById('f-export-url').value.trim();
        if (url) {
          var base = url.replace(/\/+$/, '');
          document.getElementById('btn-export-plugins-dl').onclick = function(){ window.open(base + '/wp-admin/plugin-install.php', '_blank'); };
          document.getElementById('btn-export-wpadmin').onclick = function(){ window.open(base + '/wp-admin', '_blank'); };
          document.getElementById('btn-export-zone').onclick = function(){ window.open(base + '/wp-admin/plugins.php', '_blank'); };
          document.getElementById('btn-export-wpress').onclick = function(){ window.open(base + '/wp-admin/admin.php?page=all-in-one-wp-migration-export', '_blank'); };
          showToast('URL exportación aplicada: ' + url);
        }
      }

          correo: document.getElementById('f-correo').value,
          password: document.getElementById('f-password').value,

          gdrive: document.getElementById('f-gdrive').value,

        try { localStorage.setItem('hostingConfig', JSON.stringify(config)); }

      function loadHostingConfig() {
        try {
          var saved = localStorage.getItem('hostingConfig');
          if (!saved) return;
          var c = JSON.parse(saved);
          if(c.dominio) document.getElementById('f-dominio').value = c.dominio;
          if(c.ip) document.getElementById('f-ip').value = c.ip;
          if(c.correo) document.getElementById('f-correo').value = c.correo;
          if(c.password) document.getElementById('f-password').value = c.password;
          if(c.urlapi) document.getElementById('f-urlapi').value = c.urlapi;
          if(c.currentuser) document.getElementById('f-currentuser').value = c.currentuser;
          if(c.dns1) { document.getElementById('f-dns1').value = c.dns1; document.getElementById('dns1-display').innerText = c.dns1; }
